use crate::instruction::Instruction;
use serde::Serialize;
use std::collections::HashMap;

const REG_COUNT: usize = 256;

#[derive(Debug, Clone)]
pub struct HeapObject {
    pub data: Vec<u8>,
    pub active: bool,
}

#[derive(Debug)]
pub struct VM {
    pub pc: usize,
    pub gas: i64,
    pub regs: [i64; REG_COUNT],
    pub program: Vec<Instruction>,
    pub heap: HashMap<u64, HeapObject>,
    pub next_heap_id: u64,

    // CMP flags
    pub flag_eq: bool,
    pub flag_gt: bool,
    pub flag_lt: bool,

    pub step: u64,
}

#[derive(Serialize)]
pub struct TraceEvent<'a> {
    step: u64,
    pc: usize,
    gas: i64,
    op: &'a str,
    regs: &'a [i64],
    heap_delta: Option<HeapDelta>,
}

#[derive(Serialize)]
pub struct HeapDelta {
    ptr: u64,
    offset: usize,
    value: u8,
}

impl VM {
    pub fn new(program: Vec<Instruction>, gas: i64) -> Self {
        Self {
            pc: 0,
            gas,
            regs: [0; REG_COUNT],
            program,
            heap: HashMap::new(),
            next_heap_id: 1,
            flag_eq: false,
            flag_gt: false,
            flag_lt: false,
            step: 0,
        }
    }

    pub fn run(&mut self) {
        while self.pc < self.program.len() {
            if self.gas <= 0 {
                self.panic("Out of Gas");
            }

            let instr = self.program[self.pc].clone();
            self.step += 1;
            self.gas -= 1;

            let mut heap_delta: Option<HeapDelta> = None;

            match instr {
                Instruction::NOP => {}

                Instruction::HALT => break,

                Instruction::PANIC(msg) => {
                    self.panic(&msg);
                }

                Instruction::DUMP => {
                    eprintln!("{:#?}", self);
                }

                Instruction::GAS(reg) => {
                    self.regs[reg as usize] = self.gas;
                }

                Instruction::ASSERT(reg) => {
                    if self.regs[reg as usize] == 0 {
                        self.panic("ASSERT failed");
                    }
                }

                Instruction::SET { reg, val } => {
                    self.regs[reg as usize] = val;
                }

                Instruction::SETS { reg, val } => {
                    let id = self.alloc(val.into_bytes());
                    self.regs[reg as usize] = id as i64;
                }

                Instruction::MOV { dest, src } => {
                    self.regs[dest as usize] = self.regs[src as usize];
                }

                Instruction::SWAP { r1, r2 } => {
                    self.regs.swap(r1 as usize, r2 as usize);
                }

                Instruction::ADD { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] + self.regs[s2 as usize];
                }

                Instruction::SUB { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] - self.regs[s2 as usize];
                }

                Instruction::MUL { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] * self.regs[s2 as usize];
                }

                Instruction::DIV { dest, s1, s2 } => {
                    let d = self.regs[s2 as usize];
                    if d == 0 {
                        self.panic("Division by zero");
                    }
                    self.regs[dest as usize] = self.regs[s1 as usize] / d;
                }

                Instruction::MOD { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] % self.regs[s2 as usize];
                }

                Instruction::AND { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] & self.regs[s2 as usize];
                }

                Instruction::OR { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] | self.regs[s2 as usize];
                }

                Instruction::XOR { dest, s1, s2 } => {
                    self.regs[dest as usize] =
                        self.regs[s1 as usize] ^ self.regs[s2 as usize];
                }

                Instruction::NOT { dest, src } => {
                    self.regs[dest as usize] = !self.regs[src as usize];
                }

                Instruction::CMP { r1, r2 } => {
                    let a = self.regs[r1 as usize];
                    let b = self.regs[r2 as usize];
                    self.flag_eq = a == b;
                    self.flag_gt = a > b;
                    self.flag_lt = a < b;
                }

                Instruction::JMP { target } => {
                    self.pc = target;
                    self.emit_trace("JMP", None);
                    continue;
                }

                Instruction::BEQ { target } => {
                    if self.flag_eq {
                        self.pc = target;
                        self.emit_trace("BEQ", None);
                        continue;
                    }
                }

                Instruction::BGT { target } => {
                    if self.flag_gt {
                        self.pc = target;
                        self.emit_trace("BGT", None);
                        continue;
                    }
                }

                Instruction::BLT { target } => {
                    if self.flag_lt {
                        self.pc = target;
                        self.emit_trace("BLT", None);
                        continue;
                    }
                }

                Instruction::NEW { dest, size } => {
                    let id = self.alloc(vec![0u8; size]);
                    self.regs[dest as usize] = id as i64;
                }

                Instruction::FREE { ptr } => {
                    let id = self.regs[ptr as usize] as u64;
                    if let Some(obj) = self.heap.get_mut(&id) {
                        obj.active = false;
                    }
                }

                Instruction::READ { dest, ptr, offset } => {
                    let id = self.regs[ptr as usize] as u64;
                    let obj = self.heap.get(&id).expect("Invalid heap ptr");
                    self.regs[dest as usize] =
                        *obj.data.get(offset).expect("OOB READ") as i64;
                }

                Instruction::WRITE { ptr, offset, val } => {
                    let id = self.regs[ptr as usize] as u64;
                    let v = self.regs[val as usize] as u8;
                    let obj = self.heap.get_mut(&id).expect("Invalid heap ptr");
                    obj.data[offset] = v;

                    heap_delta = Some(HeapDelta {
                        ptr: id,
                        offset,
                        value: v,
                    });
                }

                Instruction::REGEX { .. } => {
                    self.panic("REGEX disabled in experiment mode");
                }

                Instruction::TEXEC { tool, .. } => {
                    match tool {
                        3 => {} // Print allowed
                        _ => self.panic("TEXEC not allowed in experiment"),
                    }
                }
            }

            self.emit_trace(instr.op_name(), heap_delta);
            self.pc += 1;
        }
    }

    fn alloc(&mut self, data: Vec<u8>) -> u64 {
        let id = self.next_heap_id;
        self.next_heap_id += 1;
        self.heap.insert(id, HeapObject { data, active: true });
        id
    }

    fn emit_trace(&self, op: &str, heap_delta: Option<HeapDelta>) {
        let evt = TraceEvent {
            step: self.step,
            pc: self.pc,
            gas: self.gas,
            op,
            regs: &self.regs,
            heap_delta,
        };
        println!("{}", serde_json::to_string(&evt).unwrap());
    }

    fn panic(&self, msg: &str) -> ! {
        eprintln!("PANIC @ step {}: {}", self.step, msg);
        std::process::exit(1);
    }
}

trait OpName {
    fn op_name(&self) -> &str;
}

impl OpName for Instruction {
    fn op_name(&self) -> &str {
        match self {
            Instruction::NOP => "NOP",
            Instruction::HALT => "HALT",
            Instruction::PANIC(_) => "PANIC",
            Instruction::DUMP => "DUMP",
            Instruction::GAS(_) => "GAS",
            Instruction::ASSERT(_) => "ASSERT",
            Instruction::SET { .. } => "SET",
            Instruction::SETS { .. } => "SETS",
            Instruction::MOV { .. } => "MOV",
            Instruction::SWAP { .. } => "SWAP",
            Instruction::ADD { .. } => "ADD",
            Instruction::SUB { .. } => "SUB",
            Instruction::MUL { .. } => "MUL",
            Instruction::DIV { .. } => "DIV",
            Instruction::MOD { .. } => "MOD",
            Instruction::AND { .. } => "AND",
            Instruction::OR { .. } => "OR",
            Instruction::XOR { .. } => "XOR",
            Instruction::NOT { .. } => "NOT",
            Instruction::CMP { .. } => "CMP",
            Instruction::JMP { .. } => "JMP",
            Instruction::BEQ { .. } => "BEQ",
            Instruction::BGT { .. } => "BGT",
            Instruction::BLT { .. } => "BLT",
            Instruction::NEW { .. } => "NEW",
            Instruction::FREE { .. } => "FREE",
            Instruction::READ { .. } => "READ",
            Instruction::WRITE { .. } => "WRITE",
            Instruction::REGEX { .. } => "REGEX",
            Instruction::TEXEC { .. } => "TEXEC",
        }
    }
}
