#[derive(Serialize, Deserialize, Debug, Clone)]
pub enum Instruction {
    // === 1. System ===
    NOP,
    HALT,
    PANIC(String),
    DUMP,
    GAS(u8),
    ASSERT(u8), // The "Self-Awareness" check

    // === 2. Registers ===
    SET { reg: u8, val: i64 },
    SETS { reg: u8, val: String }, // String Literal support
    MOV { dest: u8, src: u8 },
    SWAP { r1: u8, r2: u8 },

    // === 3. Arithmetic ===
    ADD { dest: u8, s1: u8, s2: u8 },
    SUB { dest: u8, s1: u8, s2: u8 },
    MUL { dest: u8, s1: u8, s2: u8 },
    DIV { dest: u8, s1: u8, s2: u8 },
    MOD { dest: u8, s1: u8, s2: u8 },

    // === 4. Logic ===
    AND { dest: u8, s1: u8, s2: u8 },
    OR  { dest: u8, s1: u8, s2: u8 },
    XOR { dest: u8, s1: u8, s2: u8 },
    NOT { dest: u8, src: u8 },

    // === 5. Control Flow ===
    CMP { r1: u8, r2: u8 },
    JMP { target: usize },
    BEQ { target: usize },
    BGT { target: usize },
    BLT { target: usize },

    // === 6. Memory ===
    NEW { dest: u8, size: usize },
    FREE { ptr: u8 },
    READ { dest: u8, ptr: u8, offset: usize },
    WRITE { ptr: u8, offset: usize, val: u8 },

    // === 7. Extensions ===
    REGEX { dest: u8, pat: u8, text: u8 },
    TEXEC { tool: u16, arg: u8, dest: u8 },
}

use crate::instruction::Instruction;
use serde::Serialize;
use std::collections::HashMap;

const REG_COUNT: usize = 256;

#[derive(Debug, Clone)]
pub struct HeapObject {
    pub data: Vec<u8>,
    pub active: bool,
}

#[derive(Debug)]
pub struct VM {
    pub pc: usize,
    pub gas: i64,
    pub regs: [i64; REG_COUNT],
    pub program: Vec<Instruction>,
    pub heap: HashMap<u64, HeapObject>,
    pub next_heap_id: u64, // Monotonic ID counter

    // CMP flags
    pub flag_eq: bool,
    pub flag_gt: bool,
    pub flag_lt: bool,

    pub step: u64,
}

#[derive(Serialize)]
pub struct TraceEvent<'a> {
    step: u64,
    pc: usize,
    gas: i64,
    op: &'a str,
    regs: &'a [i64],
    heap_delta: Option<HeapDelta>,
}

#[derive(Serialize)]
pub struct HeapDelta {
    ptr: u64,
    offset: usize,
    value: u8,
}

impl VM {
    pub fn new(program: Vec<Instruction>, gas: i64) -> Self {
        Self {
            pc: 0,
            gas,
            regs: [0; REG_COUNT],
            program,
            heap: HashMap::new(),
            next_heap_id: 1, // Start from 1, 0 can be null
            flag_eq: false,
            flag_gt: false,
            flag_lt: false,
            step: 0,
        }
    }

    pub fn run(&mut self) {
        while self.pc < self.program.len() {
            // Gas Check (The Cost of Thinking)
            if self.gas <= 0 {
                self.panic("Out of Gas");
            }

            let instr = self.program[self.pc].clone();
            self.step += 1;
            self.gas -= 1;

            let mut heap_delta: Option<HeapDelta> = None;

            match instr {
                Instruction::NOP => {}
                Instruction::HALT => break,
                Instruction::PANIC(msg) => self.panic(&msg),
                Instruction::DUMP => eprintln!("{:#?}", self),
                Instruction::GAS(reg) => self.regs[reg as usize] = self.gas,
                
                // === The Killer Feature: ASSERT ===
                Instruction::ASSERT(reg) => {
                    if self.regs[reg as usize] == 0 {
                        self.panic("ASSERT failed: Agent self-check rejected state");
                    }
                }

                Instruction::SET { reg, val } => self.regs[reg as usize] = val,
                Instruction::SETS { reg, val } => {
                    let id = self.alloc(val.into_bytes());
                    self.regs[reg as usize] = id as i64;
                }

                Instruction::MOV { dest, src } => self.regs[dest as usize] = self.regs[src as usize],
                Instruction::SWAP { r1, r2 } => self.regs.swap(r1 as usize, r2 as usize),

                Instruction::ADD { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize].wrapping_add(self.regs[s2 as usize]),
                Instruction::SUB { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize].wrapping_sub(self.regs[s2 as usize]),
                Instruction::MUL { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize].wrapping_mul(self.regs[s2 as usize]),
                Instruction::DIV { dest, s1, s2 } => {
                    let d = self.regs[s2 as usize];
                    if d == 0 { self.panic("Division by zero"); }
                    self.regs[dest as usize] = self.regs[s1 as usize] / d;
                }
                Instruction::MOD { dest, s1, s2 } => {
                    let d = self.regs[s2 as usize];
                    if d == 0 { self.panic("Modulo by zero"); }
                    self.regs[dest as usize] = self.regs[s1 as usize] % d;
                }

                Instruction::AND { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize] & self.regs[s2 as usize],
                Instruction::OR { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize] | self.regs[s2 as usize],
                Instruction::XOR { dest, s1, s2 } => self.regs[dest as usize] = self.regs[s1 as usize] ^ self.regs[s2 as usize],
                Instruction::NOT { dest, src } => self.regs[dest as usize] = !self.regs[src as usize],

                Instruction::CMP { r1, r2 } => {
                    let a = self.regs[r1 as usize];
                    let b = self.regs[r2 as usize];
                    self.flag_eq = a == b;
                    self.flag_gt = a > b;
                    self.flag_lt = a < b;
                }

                Instruction::JMP { target } => { self.pc = target; self.emit_trace("JMP", None); continue; }
                Instruction::BEQ { target } => { if self.flag_eq { self.pc = target; self.emit_trace("BEQ", None); continue; } }
                Instruction::BGT { target } => { if self.flag_gt { self.pc = target; self.emit_trace("BGT", None); continue; } }
                Instruction::BLT { target } => { if self.flag_lt { self.pc = target; self.emit_trace("BLT", None); continue; } }

                Instruction::NEW { dest, size } => {
                    let id = self.alloc(vec![0u8; size]);
                    self.regs[dest as usize] = id as i64;
                }

                Instruction::FREE { ptr } => {
                    let id = self.regs[ptr as usize] as u64;
                    if let Some(obj) = self.heap.get_mut(&id) {
                        obj.active = false;
                    }
                }

                // === Safe Memory Access with Soft Panic ===
                Instruction::READ { dest, ptr, offset } => {
                    let id = self.regs[ptr as usize] as u64;
                    match self.heap.get(&id) {
                        Some(obj) => {
                            if !obj.active { self.panic(&format!("Segfault: Read from freed ptr {}", id)); }
                            if offset >= obj.data.len() { self.panic(&format!("Segfault: OOB Read at offset {} (len {})", offset, obj.data.len())); }
                            self.regs[dest as usize] = obj.data[offset] as i64;
                        },
                        None => self.panic(&format!("Segfault: Invalid ptr {}", id)),
                    }
                }

                Instruction::WRITE { ptr, offset, val } => {
                    let id = self.regs[ptr as usize] as u64;
                    let v = self.regs[val as usize] as u8;
                    match self.heap.get_mut(&id) {
                        Some(obj) => {
                            if !obj.active { self.panic(&format!("Segfault: Write to freed ptr {}", id)); }
                            if offset >= obj.data.len() { self.panic(&format!("Segfault: OOB Write at offset {} (len {})", offset, obj.data.len())); }
                            obj.data[offset] = v;
                            heap_delta = Some(HeapDelta { ptr: id, offset, value: v });
                        },
                        None => self.panic(&format!("Segfault: Invalid ptr {}", id)),
                    }
                }

                // === Experiment Constraints ===
                Instruction::REGEX { .. } => self.panic("REGEX disabled in experiment mode"),
                Instruction::TEXEC { tool, arg: _, dest: _ } => {
                    match tool {
                        3 => { /* Print is allowed, maybe log to stderr or a special buffer */ } 
                        _ => self.panic("TEXEC blocked: Sandbox violation"),
                    }
                }
            }

            self.emit_trace(instr.op_name(), heap_delta);
            self.pc += 1;
        }
    }

    fn alloc(&mut self, data: Vec<u8>) -> u64 {
        let id = self.next_heap_id;
        self.next_heap_id += 1; // Monotonic increase
        self.heap.insert(id, HeapObject { data, active: true });
        id
    }

    fn emit_trace(&self, op: &str, heap_delta: Option<HeapDelta>) {
        let evt = TraceEvent {
            step: self.step,
            pc: self.pc,
            gas: self.gas,
            op,
            regs: &self.regs,
            heap_delta,
        };
        // Use println! for trace logs, standard output for Python to capture
        println!("{}", serde_json::to_string(&evt).unwrap());
    }

    fn panic(&self, msg: &str) -> ! {
        // Log the final panic as a special JSON or stderr
        eprintln!("PANIC @ step {}: {}", self.step, msg);
        std::process::exit(1);
    }
}

Text-to-Bytecode 编译器。支持 Label 和 .l0 语法。
use crate::instruction::Instruction;
use std::collections::HashMap;

pub fn compile_source(source: &str) -> Result<Vec<Instruction>, String> {
    let mut instructions = Vec::new();
    let mut labels = HashMap::new();
    let mut pending_jumps = Vec::new(); // (instruction_index, label_name)

    // Pass 1: Parse instructions and record label positions
    let mut pc = 0;
    
    // Simple line-based parser
    for (line_num, line) in source.lines().enumerate() {
        let line = line.trim();
        let code = line.split(';').next().unwrap_or("").trim(); // Remove comments
        if code.is_empty() { continue; }

        if code.ends_with(':') {
            let label = code.trim_end_matches(':').trim();
            if labels.contains_key(label) {
                return Err(format!("Line {}: Duplicate label '{}'", line_num, label));
            }
            labels.insert(label.to_string(), pc);
            continue;
        }

        // Parse Instruction
        // Split by whitespace but respect quotes for SETS
        let parts = parse_parts(code);
        if parts.is_empty() { continue; }

        let op = parts[0].to_uppercase();
        let args = &parts[1..];

        let instr = match op.as_str() {
            "NOP" => Instruction::NOP,
            "HALT" => Instruction::HALT,
            "PANIC" => Instruction::PANIC(args.get(0).unwrap_or(&"User Panic".to_string()).to_string()),
            "DUMP" => Instruction::DUMP,
            "GAS" => Instruction::GAS(parse_reg(args[0])?),
            "ASSERT" => Instruction::ASSERT(parse_reg(args[0])?),
            
            "SET" => Instruction::SET { reg: parse_reg(args[0])?, val: parse_int(args[1])? },
            "SETS" => Instruction::SETS { reg: parse_reg(args[0])?, val: args[1].trim_matches('"').to_string() },
            "MOV" => Instruction::MOV { dest: parse_reg(args[0])?, src: parse_reg(args[1])? },
            "SWAP" => Instruction::SWAP { r1: parse_reg(args[0])?, r2: parse_reg(args[1])? },
            
            "ADD" => Instruction::ADD { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "SUB" => Instruction::SUB { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "MUL" => Instruction::MUL { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "DIV" => Instruction::DIV { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "MOD" => Instruction::MOD { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            
            "AND" => Instruction::AND { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "OR"  => Instruction::OR  { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "XOR" => Instruction::XOR { dest: parse_reg(args[0])?, s1: parse_reg(args[1])?, s2: parse_reg(args[2])? },
            "NOT" => Instruction::NOT { dest: parse_reg(args[0])?, src: parse_reg(args[1])? },
            
            "CMP" => Instruction::CMP { r1: parse_reg(args[0])?, r2: parse_reg(args[1])? },
            
            "JMP" => { pending_jumps.push((pc, args[0].to_string())); Instruction::JMP { target: 0 } },
            "BEQ" => { pending_jumps.push((pc, args[0].to_string())); Instruction::BEQ { target: 0 } },
            "BGT" => { pending_jumps.push((pc, args[0].to_string())); Instruction::BGT { target: 0 } },
            "BLT" => { pending_jumps.push((pc, args[0].to_string())); Instruction::BLT { target: 0 } },
            
            "NEW" => Instruction::NEW { dest: parse_reg(args[0])?, size: parse_usize(args[1])? },
            "FREE" => Instruction::FREE { ptr: parse_reg(args[0])? },
            "READ" => Instruction::READ { dest: parse_reg(args[0])?, ptr: parse_reg(args[1])?, offset: parse_usize(args[2])? },
            "WRITE" => Instruction::WRITE { ptr: parse_reg(args[0])?, offset: parse_usize(args[1])?, val: parse_reg(args[2])? },
            
            // Experiment Stubs
            "REGEX" => Instruction::REGEX { dest: 0, pat: 0, text: 0 }, 
            "TEXEC" => Instruction::TEXEC { tool: 3, arg: parse_reg(args[0])?, dest: 0 }, // Simplified for Print

            _ => return Err(format!("Unknown opcode: {}", op)),
        };
        
        instructions.push(instr);
        pc += 1;
    }

    // Pass 2: Resolve Jumps
    for (idx, label) in pending_jumps {
        let target = *labels.get(&label).ok_or(format!("Undefined label: {}", label))?;
        match &mut instructions[idx] {
            Instruction::JMP { target: t } => *t = target,
            Instruction::BEQ { target: t } => *t = target,
            Instruction::BGT { target: t } => *t = target,
            Instruction::BLT { target: t } => *t = target,
            _ => {}
        }
    }

    Ok(instructions)
}

fn parse_reg(s: &str) -> Result<u8, String> {
    let s = s.trim().trim_end_matches(',');
    let s = s.strip_prefix('R').or_else(|| s.strip_prefix('r')).unwrap_or(s);
    s.parse::<u8>().map_err(|_| format!("Invalid reg: {}", s))
}

fn parse_int(s: &str) -> Result<i64, String> {
    let s = s.trim().trim_end_matches(',');
    s.parse::<i64>().map_err(|_| format!("Invalid int: {}", s))
}

fn parse_usize(s: &str) -> Result<usize, String> {
    let s = s.trim().trim_end_matches(',');
    s.parse::<usize>().map_err(|_| format!("Invalid usize: {}", s))
}

fn parse_parts(code: &str) -> Vec<String> {
    // Basic tokenizer that handles quoted strings roughly
    let mut parts = Vec::new();
    let mut current = String::new();
    let mut in_quote = false;
    
    for c in code.chars() {
        if c == '"' {
            in_quote = !in_quote;
            current.push(c);
        } else if c.is_whitespace() && !in_quote {
            if !current.is_empty() {
                parts.push(current.clone());
                current.clear();
            }
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        parts.push(current);
    }
    parts
}

use clap::{Parser, Subcommand};
use std::fs::File;
use std::io::{BufReader, BufWriter};
use std::path::PathBuf;
use crate::vm::VM;

#[derive(Parser)]
#[command(name = "lzero")]
#[command(version = "0.2-frozen")]
#[command(about = "L-Zero: Deterministic Agent Runtime")]
struct Cli {
    #[command(subcommand)]
    command: Commands,
}

#[derive(Subcommand)]
enum Commands {
    /// Compile .l0 source code to binary bytecode
    Build {
        source: PathBuf,
        #[arg(short, long)]
        output: PathBuf,
    },
    /// Run binary bytecode
    Run {
        file: PathBuf,
        #[arg(short, long, default_value_t = 1000)]
        gas: i64,
    },
}

fn main() {
    let cli = Cli::parse();

    match cli.command {
        Commands::Build { source, output } => {
            let content = std::fs::read_to_string(&source).expect("Failed to read source");
            match assembler::compile_source(&content) {
                Ok(program) => {
                    let file = File::create(&output).expect("Failed to create output file");
                    let mut writer = BufWriter::new(file);
                    bincode::serialize_into(&mut writer, &program).expect("Serialization failed");
                    eprintln!("✅ Built {:?} -> {:?}", source, output);
                }
                Err(e) => {
                    eprintln!("❌ Compilation Error: {}", e);
                    std::process::exit(1);
                }
            }
        }
        Commands::Run { file, gas } => {
            let f = File::open(&file).expect("Failed to open file");
            let mut reader = BufReader::new(f);
            match bincode::deserialize_from(&mut reader) {
                Ok(program) => {
                    let mut vm = VM::new(program, gas);
                    vm.run();
                }
                Err(e) => {
                    eprintln!("❌ Execution Error: Invalid Bytecode ({})", e);
                    std::process::exit(1);
                }
            }
        }
    }
}
